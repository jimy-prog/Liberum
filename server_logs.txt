#!/bin/bash
cd /home/ubuntu/liberum
git fetch origin main
LOCAL=$(git rev-parse HEAD)
REMOTE=$(git rev-parse origin/main)

if [ $LOCAL != $REMOTE ]; then
    echo "$(date) - Changes detected. Pulling..." >> /home/ubuntu/auto_deploy.log
    git pull origin main
    echo "$(date) - Rebuilding docker containers..." >> /home/ubuntu/auto_deploy.log
    cd /home/ubuntu/liberum-data
    docker-compose down
    docker-compose up -d --build
    echo "$(date) - Deployment complete." >> /home/ubuntu/auto_deploy.log
fi
